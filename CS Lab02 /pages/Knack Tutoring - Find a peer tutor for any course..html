"""

Function Name: findStops()

Parameters: budget (int), minTime (float)

Returns: itinerary (list)

"""


def findStops(budget, minTime):

  num_places = 0

  total_time = minTime

  itinerary = []

  try:

    file = open("stopsData.txt", "r")

    raw_data = file.readlines()

    file.close()

  except:

    raw_data = []


  lines = []

  for line in raw_data:

    filter_line = line.strip()

    if filter_line != "":

      lines.append(filter_line)


  i = 0

  while i < len(lines):

    if i + 2 >= len(lines):

      break


    name = lines[i]

    cost = float(lines[i + 1].replace("$",""))

    time = float(lines[i + 2])

    if cost <= budget:

      itinerary.append(name)

      budget -= cost

      total_time += round(time / 60.0, 2)

      num_places += 1

    i += 3


  total_time = round(total_time, 2)

  if num_places >= 1:

    print(f"We stopped at {num_places} places, and the trip took {total_time} hours.")

  else:

    print("We couldn't afford to stop anywhere.")

  return itinerary

Returns: None (NoneType)

def bobaOrCoffee(budget, groupSize):

  try:

    files = open("bobaVsCoffeePicks.txt", "r")

    raw_data = files.readlines()

    files.close()

  except:

    raw_data = []

  lines = []

  for line in raw_data:

    filter_line = line.strip()

    if filter_line != "":

      lines.append(filter_line)


  totalRating = 0.0

  shopList = []

  i = 0

  while i < len(lines):

    if i + 5 >= len(lines):

      break

    coffeeName = lines[i].strip()

    coffeePrice = lines[i + 1].strip()

    coffeeRating = lines[i + 2].strip()

    city1, shop1 = [x.strip() for x in coffeeName.split("|")]


    price1 = float(coffeePrice.replace("$", ""))

    rating1 = float(coffeeRating)


    bobaName = lines[i + 3].strip()

    bobaPrice = lines[i + 4].strip()

    bobaRating = lines[i + 5].strip()

    city2, shop2 = [x.strip() for x in bobaName.split("|")]


    price2 = float(bobaPrice.replace("$",""))

    rating2 = float(bobaRating)

    totalCost1 = round(price1 * groupSize, 2)

    totalCost2 = round(price2 * groupSize, 2)


    affordable1 = budget >= totalCost1

    affordable2 = budget >= totalCost2

    chosenShop = ""

    chosenRating = 0.0

    chosenCost = 0.0


    if (affordable1 and affordable2):

      if price1 == price2:

        if rating1 >= rating2:

          chosenShop = shop1

          chosenRating = rating1

          chosenCost = totalCost1

        else:

          chosenShop = shop2

          chosenRating = rating2

          chosenCost = totalCost2

      elif price1 > price2:

        chosenShop = shop2

        chosenRating = rating2

        chosenCost = totalCost2

      else:

        chosenShop = shop1

        chosenRating = rating1

        chosenCost = totalCost1

    elif (affordable1):

      chosenShop = shop1

      chosenRating = rating1

      chosenCost = totalCost1

    elif (affordable2):

      chosenShop = shop2

      chosenRating = rating2

      chosenCost = totalCost2


    if chosenShop != "":

      budget -= chosenCost

      shopList.append((city1, chosenShop))

      totalRating += chosenRating

    i += 6


  outfile = open("bestChoices.txt", "w")

  if len(shopList) == 0:

    outfile.write("Plan does not work out, maybe next time...")

  else:

    outfile.write("Make sure to visit each place we picked:\n\n")

    for city, shop in shopList:

      outfile.write(f"- In {city}, we would stop by {shop}\n")

    averageRating = round(totalRating / len(shopList), 1)

    outfile.write(f"\nThe average rating of all shops is {averageRating}. So excited!")

  outfile.close()

"""

Function Name: csvToList()

Parameters: fileName (str)

Returns: flightsList (list)

"""


def csvToList(fileName):

  flights = []

  try:

    with open(fileName, "r") as file:

      reader = csv.reader(file)

      next(reader, None)

      for row in reader:

        if row:

          origin = row[0]

          destination = row[1]

          emissions = int(row[2])

          flight_tuple = (origin, destination, emissions)

          flights.append(flight_tuple)

      return flights

  except FileNotFoundError:

    return []

"""

Function Name: csvToDict()

Parameters: fileName (str)

Returns: flightsDict (dict)

"""


def csvToDict(fileName):

  newDict = {}

  try:

    with open(fileName, "r") as file:

      reader = csv.reader(file)

      next(reader, None)

      for row in reader:

        if row:

          origin = row[0]

          destination = row[1]

          emissions = int(row[2])

          key = (origin, destination)

          if key not in newDict:

            newDict[key] = [emissions]

          else:

            newDict[key].append(emissions)

  except FileNotFoundError:

    return {}

  return newDict

"""

Function Name: worstRoutes_list()

Parameters: originCountry (str), destCountry (str), flightsList (list)

Returns: message (str)

"""


def worstRoutes_list(originCountry: str, destCountry: str, flightsList: list):

  worst_emissions = -1

  for origin, destination, emissions in flightsList:

    if origin == originCountry and destination == destCountry:

      if emissions > worst_emissions:

        worst_emissions = emissions

  return f"{originCountry} to {destCountry}: {worst_emissions} carbon emissions."

"""

Function Name: worstRoutes_dict()

Parameters: originCountry (str), destCountry (str), flightsDict (dict)

Returns: message (str)

"""


def worstRoutes_dict(originCountry: str, destCountry: str, flightsDict: dict):

  key = (originCountry, destCountry)

  emissions_list = flightsDict[key]

  worst_emissions = max(emissions_list)

  return f"{originCountry} to {destCountry}: {worst_emissions} carbon emissions."
